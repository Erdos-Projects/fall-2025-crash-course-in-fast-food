\section{Data Overview}
\begin{frame}{From Bites to Bytes: Building Our Dataset}
    \begin{block}{Source for Fast-Food}
        \begin{itemize}
            \item Open Street Map \& Overpass API
            \item Extraction of (fast-food) coordinates and road features\\ 
            (e.g.~traffic lights, stop signs)
            \item Format: name/item \& coordinates for different subcategories of places
            \begin{itemize}
                \item Food-through-window Drive-thrus
                \item Curbside pickup
            \end{itemize}
            
            %\item Issue: 'drive-through' tag often 'None' instead of 'yes/no'
            
        \end{itemize}
    \end{block}
    
    \begin{block}{Source for Crashes}
        \begin{itemize}
            \item Public records request to  Ohio Department of Public Safety
            \begin{itemize}
                \item csv file: 1.289.913 crashes in Ohio between Sep 2019 and Sep 2025 \\(in Franklin County: 115.682)
                \item Most relevant columns: coordinates and ''Narrative'' (written police report) $\rightarrow$ is a crash associated with food?
            \end{itemize}
            
      
        \end{itemize}
    \end{block}
    
\end{frame}
