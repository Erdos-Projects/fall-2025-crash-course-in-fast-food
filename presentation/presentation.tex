\section{Data Overview}
\begin{frame}{From Bites to Bytes: Building Our Dataset}
    \begin{block}{Source for Fast-Food locations}
        \begin{itemize}
            \item Open Street Map API
            \item 6153 locations in Ohio
            \item Issue: 'drive-through' tag often 'None' instead of 'yes/no'
            
        \end{itemize}
    \end{block}
    
    \begin{block}{Source for Crashes}
        \begin{itemize}
            \item Public records request to  Ohio Department of Public Safety
            \item Period: Sep 2019--Sep 2025 (5 full years)
            \item Total number of crashes: 1.289.913
            \item Main filter: explicit mention of fast-food place in the crash report narrative (32.760)
        \end{itemize}
    \end{block}
    
\end{frame}


\begin{frame}{Fast-food location filtering and things to compare}
    Our subsets to compare and the reasoning behind it:
    \begin{itemize}
        \item Top 20 non-pizza fast food chains
        \begin{itemize}
            \item In queue, food is handed through window
            \item Stressful situation to handle food and driving away at the same time, very distracted
        \end{itemize}
        \item Top X pizza chains
        \begin{itemize}
            \item Food is brought to the curb
            \item No pressure of having to make space for someone immideatly, more relaxed
            \item Less distraction
        \end{itemize} 
        \item Top Y coffee chains
        \begin{itemize}
            \item Does the need for coffeine play a role?
        \end{itemize}
        \item Top Z gas stations
        \begin{itemize}
            \item similar road situation, but less distracted due to food
        \end{itemize}
    \end{itemize}
    
\end{frame}
